$ErrorActionPreference = "Stop"

$SCRIPT_PATH = $PSCommandPath

if ([string]::IsNullOrWhiteSpace($SCRIPT_PATH) -and $MyInvocation.MyCommand.Path) {
  $SCRIPT_PATH = $MyInvocation.MyCommand.Path
}

$PROJECT_DIR = if ([string]::IsNullOrWhiteSpace($SCRIPT_PATH)) {
  (Get-Location).Path
} else {
  Split-Path -Parent $SCRIPT_PATH
}
$BRANCH = "main"
$COMMIT_MESSAGE = "푝 요청사항 반영"

Set-Location $PROJECT_DIR

$ignoreEntries = @(
  ".env.backup-*",
  "*.bak-*",
  "*.bak.*",
  "discord-bot-vm.tar.gz",
  "discord-bot-vm-deploy.sh",
  "preview-*.png",
  "preview-*.svg"
)

if (-not (Test-Path -LiteralPath ".gitignore")) {
  New-Item -Path ".gitignore" -ItemType File | Out-Null
}

foreach ($entry in $ignoreEntries) {
  $exists = Select-String `
    -Path ".gitignore" `
    -SimpleMatch `
    -Pattern $entry `
    -Quiet `
    -ErrorAction SilentlyContinue

  if (-not $exists) {
    Add-Content -Path ".gitignore" -Value $entry
  }
}

git status --short

git add -- `
  src `
  test `
  assets `
  scripts `
  package.json `
  package-lock.json `
  requirements-chess.txt `
  ecosystem.config.cjs `
  .env.example `
  VM_DEPLOY.md `
  README.md `
  .gitignore `
  deploy-vm.ps1 `
  그긴거.md

git diff --cached --quiet

if ($LASTEXITCODE -eq 0) {
  Write-Host "GitHub에 올릴 변경사항이 없습니다."
} elseif ($LASTEXITCODE -eq 1) {
  git commit -m $COMMIT_MESSAGE
  git push origin $BRANCH
  git log -1 --oneline
} else {
  throw "git diff 확인 중 오류가 발생했습니다."
}
