$ErrorActionPreference = "Stop"

$PROJECT_DIR = "C:\Users\MSI\OneDrive\바탕 화면\디코봇\discord"
$BRANCH = "main"

Set-Location $PROJECT_DIR

$ignoreEntries = @(
  ".env.backup-*",
  "*.bak-*",
  "*.bak.*"
)

if (-not (Test-Path ".gitignore")) {
  New-Item ".gitignore" -ItemType File | Out-Null
}

foreach ($entry in $ignoreEntries) {
  $exists = Select-String `
    -Path ".gitignore" `
    -SimpleMatch `
    -Pattern $entry `
    -Quiet

  if (-not $exists) {
    Add-Content ".gitignore" $entry
  }
}

git status --short

git add -- `
  src `
  test `
  assets `
  scripts `
  package.json `
  package-lock.json `
  requirements-chess.txt `
  ecosystem.config.cjs `
  .env.example `
  VM_DEPLOY.md `
  README.md `
  .gitignore `
  deploy-vm.ps1 `
  그긴거.md `

git diff --cached --quiet

if ($LASTEXITCODE -eq 0) {
  Write-Host "Git에 올릴 변경사항이 없습니다."
} elseif ($LASTEXITCODE -eq 1) {
  git commit -m "먼치업데이트"
  git push origin $BRANCH
  git log -1 --oneline
} else {
  throw "git diff 확인 중 오류가 발생했습니다."
}
